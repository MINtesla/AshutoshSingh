TrafficSegregationService.java
@Service
@Slf4j
public class TrafficSegregationService {

  @Autowired
  private MsisdnCacheService cacheService;
  @Autowired
  private CuigForwarderService forwarder;

  public CompletableFuture<RoutingDecision>
        route(TrafficEvent event) {

    return CompletableFuture
      .supplyAsync(() ->
          cacheService.lookup(event.getMsisdn()))
      .thenApply(sub -> evaluate(event, sub))
      .thenCompose(forwarder::dispatch)
      .exceptionally(ex -> {
        log.error("routing failed", ex);
        return RoutingDecision.fallback();
      });
  }
}
